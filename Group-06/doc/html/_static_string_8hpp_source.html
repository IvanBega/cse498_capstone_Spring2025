1167 void trim() {
1168 std::size_t start = 0;
1169 std::size_t end = mCurrentSize;
1170
1171 // Trim leading whitespace
1172 while (start < mCurrentSize &&
1173 (mString[start] == ' ' || mString[start] == '\t' ||
1174 mString[start] == '\r')) {
1175 ++start;
1176 }
1177
1178 // Trim trailing whitespace
1179 while (end > start &&
1180 (mString[end - 1] == ' ' || mString[end - 1] == '\t' ||
1181 mString[end - 1] == '\r')) {
1182 --end;
1183 }
1184
1185 if (start > 0 || end < mCurrentSize) {
1186 // Remove the trailing or leading white spaces from the string
1187 std::memmove(mString, mString + start, end - start);
1188 mCurrentSize = end - start;
1189 mString[mCurrentSize] = null_terminator;
1190
1191 // Clear the remaining part of the string, make sure properly handled
1192 for (std::size_t i = mCurrentSize + 1; i <= MaxSize; ++i) {
1193 mString[i] = null_terminator;
1194 }
