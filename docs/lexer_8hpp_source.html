313 // Generate and return the next token from the input stream.
314 Token NextToken(std::string_view in) {
315 // If we cannot read in, return an "EOF" token.
316 if (start_pos >= std::ssize(in)) return { 0, "", cur_line, cur_col };
317
318 int cur_pos = start_pos; // Position in the input that we are actively analyzing
319 int best_pos = start_pos; // Best look-ahead we've found so far
320 int cur_state = 0; // Next state for the DFA analysis
321 int cur_stop = 0; // Current "stop" state (or 0 if we can't stop here)
322 int best_stop = -1; // Best stop state found so far (longest match w/ max stop value)
323
324 // If we are at the START OF A LINE, send a DFA::SYMBOL_START
325 if (start_pos == 0 || in[start_pos-1] == '\n') {
326 cur_state = DFA::GetNext(0, DFA::SYMBOL_START);
327 }
328 // Keep looking as long as:
329 // 1: We may be able to continue the current lexeme, and
330 // 2: We have not entered an invalid state, and
331 // 3: Our input string has more symbols to provide
332 while (cur_stop >= 0 && cur_state >= 0 && cur_pos < std::ssize(in)) {
333 const char next_char = in[cur_pos++];
334 if (next_char < 0) break; // Ignore invalid chars.
335 cur_state = DFA::GetNext(cur_state, next_char);
336 cur_stop = DFA::GetStop(cur_state);
337 // If we found a valid stopping point, save it as a new best.
338 if (cur_stop > 0) { best_pos = cur_pos; best_stop = cur_stop; }
339 // Look ahead to see if we are at the END OF A LINE that can finish a token.
340 if (cur_pos == std::ssize(in) || in[cur_pos] == '\n') {
341 int eol_state = DFA::GetNext(cur_state, DFA::SYMBOL_STOP);
342 int eol_stop = DFA::GetStop(eol_state);
343 if (eol_stop > 0) { best_pos = cur_pos; best_stop = eol_stop; }
344 }
345 }
346
347 // If we did not find any options, peel off just one character and use it as id.
348 if (best_pos == start_pos) { best_stop=in[start_pos]; best_pos++;}
349
350 lexeme = in.substr(start_pos, best_pos-start_pos);
351 start_pos += std::ssize(lexeme);
352
353 // Update the line number we are on.
354 const size_t out_line = cur_line;
355 const size_t out_col = cur_col;
356 if ((cur_col = lexeme.rfind('\n')) == std::string::npos) { // No newlines
357 cur_col = out_col + lexeme.size();
358 } else {
359 cur_col = lexeme.size() - cur_col - 1;
360 cur_line += static_cast<size_t>(std::count(lexeme.begin(),lexeme.end(),'\n'));
361 }
362
363 // Return the token we found.
364 return { best_stop, lexeme, out_line, out_col };
365 }

367 // Convert an input string into a vector of tokens.
368 const std::vector<Token> & Tokenize(std::string_view in) {
369 start_pos = 0; // Start processing at beginning of string.
370 cur_line = 1; // Start processing at the first line of the input.
371 cur_col = 0; // Start processing at the first position of the input.
372 tokens.resize(0);
373 token_id = 0;
374 while (Token token = NextToken(in)) {
375 if (!IgnoreToken(token.id)) tokens.push_back(token);
376 }
377 return tokens;
378 }
