363template <typename T>
364bool cse::StringSet<T>::is_subset(const StringSet<T> &other) const {
365 // Prepare a lambda that checks if captured element is in the other set
366 // Used DeepSeek to assist with this lambda
367 auto otherPredicate = [&other](const T &element) {
368 return other.count(element) == 1;
369 };

371 // use all_of in ranges to loop through each element in mElements
372 return std::ranges::all_of(mElements, otherPredicate);

379template <typename T>
380void cse::StringSet<T>::substring_filter(const T &substring) {
381 // this function was written with the help from DeepSeek
382 if (substring.empty()) return; // do not do anything for an empty substring
383
384 // Lambda that will be used in RemoveFilter to filter words that we need
385 auto substringPredicate = [substring](const T &element) {
386 std::string_view element_view(element);
387 std::string_view substr_view(substring);
388 return element_view.find(substr_view) != std::string_view::npos;
389 };
390
391 remove_filter(substringPredicate);

470template <typename T>
471cse::StringSet<T> cse::StringSet<T>::random_sample(
472 int sampleSize, const std::function<bool(const T &)> filter) {
473 StringSet<T> result;
474 if (sampleSize <= 0) {
475 return result;
476 }
477
478 std::vector<T> elements;
479 for (const auto &element : mElements) {
480 if (filter(element)) {
481 elements.push_back(element);
482 }
483 }
484
485 if (static_cast<std::unordered_set<T>::size_type>(sampleSize) >
486 elements.size()) {
487 sampleSize = elements.size();
488 }
489
490 std::random_device rd;
491 std::mt19937 gen(rd());
492 std::shuffle(elements.begin(), elements.end(), gen);
493
494 for (int i = 0; i < sampleSize; ++i) {
495 result.insert(elements[i]);
496 }
497
498 return result;
499}

505template <typename T>
506std::tuple<int, double, T, T> cse::StringSet<T>::statistics() {
507 int numStrings = size();
508 if (numStrings == 0) {
509 return std::make_tuple(0, 0.0, "", "");
510 }
511
512 int totalLength = 0;
513 T longestString = "";
514 T shortestString = *mElements.begin();
515
516 for (const auto &str : mElements) {
517 int length = static_cast<int>(str.length());
518 totalLength += length;
519 if (length > static_cast<int>(longestString.length())) {
520 longestString = str;
521 }
522 if (length < static_cast<int>(shortestString.length())) {
523 shortestString = str;
524 }
525 }
526
527 double avgLength = static_cast<double>(totalLength) / numStrings;
528
529 return std::make_tuple(numStrings, avgLength, longestString, shortestString);
530}
