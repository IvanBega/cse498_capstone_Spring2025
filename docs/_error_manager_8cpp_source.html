105 void ErrorManager::logToFile(std::string& output) {
106 if (mLogFile.is_open()) {
107 if (mColorsEnabled) {
108 // Remove ANSI color codes for output in the file
109 // Tried to use RegEx, as suggested by a reviewer,
110 // but it was too difficult to implement
111
112 size_t pos = 0;
113 while ((pos = output.find("\033[")) != std::string::npos) {
114 size_t endPos = output.find('m', pos);
115 if (endPos != std::string::npos) {
116 output.erase(pos, endPos - pos + 1); // erase color code
117 }
118 }
119
120 mLogFile << output;
121 mLogFile.flush();
